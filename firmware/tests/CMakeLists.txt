cmake_minimum_required(VERSION 3.20)
project(palpable_firmware_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(FIRMWARE "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(CJSON_SOURCE_DIR "" CACHE PATH "Existing cJSON source directory for offline tests")
if(NOT CJSON_SOURCE_DIR)
    include(FetchContent)
    file(READ "${FIRMWARE}/upstream.lock.json" LOCK)
    string(JSON CJSON_REPO GET "${LOCK}" host_cjson_repository)
    string(JSON CJSON_REV GET "${LOCK}" host_cjson_revision)
    # Only populate sources; the small static library is configured below.
    FetchContent_Declare(cjson_source GIT_REPOSITORY "${CJSON_REPO}" GIT_TAG "${CJSON_REV}"
        SOURCE_SUBDIR palpable-no-subproject)
    FetchContent_MakeAvailable(cjson_source)
    set(CJSON_SOURCE_DIR "${cjson_source_SOURCE_DIR}")
endif()
add_library(cjson STATIC "${CJSON_SOURCE_DIR}/cJSON.c")
target_include_directories(cjson PUBLIC "${CJSON_SOURCE_DIR}")
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/palpable_schema.h"
    COMMAND "${Python3_EXECUTABLE}" "${FIRMWARE}/scripts/gen_schema.py" "${CMAKE_CURRENT_BINARY_DIR}/palpable_schema.h"
    DEPENDS "${FIRMWARE}/scripts/gen_schema.py" "${FIRMWARE}/../protocol/palpable.schema.json")
add_library(palpable STATIC "${FIRMWARE}/main/palpable_client.cc" "${CMAKE_CURRENT_BINARY_DIR}/palpable_schema.h")
target_include_directories(palpable PUBLIC "${FIRMWARE}/main" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(palpable PUBLIC cjson)
target_compile_options(palpable PRIVATE -Wall -Wextra -Werror)
add_executable(client_test client_test.cc)
target_link_libraries(client_test PRIVATE palpable)
target_compile_options(client_test PRIVATE -Wall -Wextra -Werror)
# assert() must remain active even when a developer chooses a release build.
target_compile_options(client_test PRIVATE -UNDEBUG)
add_executable(gateway_probe gateway_probe.cc)
target_link_libraries(gateway_probe PRIVATE palpable)
enable_testing()
add_test(NAME client_contract COMMAND client_test "${FIRMWARE}/../protocol/examples")
